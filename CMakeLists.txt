cmake_minimum_required(VERSION 3.16)
project(RetroChess VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets Concurrent)
if(MSVC)
 add_compile_options(/utf-8)
endif()
add_library(chess_core src/chess.cpp src/search.cpp)
target_include_directories(chess_core PUBLIC src)
add_library(retro_ui src/boardwidget.cpp src/mainwindow.cpp src/enginebridge.cpp)
target_link_libraries(retro_ui PUBLIC chess_core Qt6::Widgets Qt6::Concurrent)
target_include_directories(retro_ui PUBLIC src)
add_executable(RetroChess WIN32 MACOSX_BUNDLE src/main.cpp)
set_target_properties(RetroChess PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.retrochess.RetroChess MACOSX_BUNDLE_BUNDLE_NAME RetroChess)
target_link_libraries(RetroChess PRIVATE retro_ui)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(engine_tests tests/engine_tests.cpp)
 target_link_libraries(engine_tests PRIVATE chess_core)
 add_test(NAME chess_rules_and_search COMMAND engine_tests)
 add_executable(uci_tests tests/uci_tests.cpp)
 target_link_libraries(uci_tests PRIVATE retro_ui)
 add_executable(ui_tests tests/ui_tests.cpp)
 target_link_libraries(ui_tests PRIVATE retro_ui Qt6::Test)
 add_test(NAME native_interface COMMAND ui_tests)
 set_tests_properties(native_interface PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS RetroChess RUNTIME DESTINATION bin BUNDLE DESTINATION .)
